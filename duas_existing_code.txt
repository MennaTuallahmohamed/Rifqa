// نموذج الدعاء - Rifqa
// يتطلب Dart 3+ (switch expressions).

enum DuaCategory { umrah, hajj, tawaf, sai, arafah, travel, mosque, general }

/// نوع المصدر. أول ثلاثة = محتوى موثّق. الباقي لازم يظهر بوضوح إنه مش نص مأثور.
enum DuaSourceType { quran, hadith, mathur, generalPermissible, userCreated, aiAssisted }

/// مرحلة المناسك. "الدعاء المناسب دلوقتي" (المرحلة 3) هيعتمد عليها.
/// جاهزة لاحقاً لإضافة الوقت/المكان من غير تغيير الموديل.
enum RitualStage {
  none,
  preTravel,
  ihram,
  enteringHaram,
  tawaf,
  afterTawaf,
  sai,
  arafah,
  muzdalifah,
  jamarat,
}

/// جاهز لربط الـBackend لاحقاً.
enum SyncStatus { localOnly, pending, synced }

extension DuaCategoryX on DuaCategory {
  String get label => switch (this) {
        DuaCategory.umrah => 'أدعية العمرة',
        DuaCategory.hajj => 'أدعية الحج',
        DuaCategory.tawaf => 'أدعية الطواف',
        DuaCategory.sai => 'أدعية السعي',
        DuaCategory.arafah => 'أدعية عرفة',
        DuaCategory.travel => 'أدعية السفر',
        DuaCategory.mosque => 'دخول المسجد والخروج',
        DuaCategory.general => 'أذكار وأدعية عامة',
      };

  String get emoji => switch (this) {
        DuaCategory.umrah => '🕋',
        DuaCategory.hajj => '🏔️',
        DuaCategory.tawaf => '🔄',
        DuaCategory.sai => '🚶',
        DuaCategory.arafah => '🤲',
        DuaCategory.travel => '✈️',
        DuaCategory.mosque => '🕌',
        DuaCategory.general => '📿',
      };
}

extension DuaSourceTypeX on DuaSourceType {
  String get label => switch (this) {
        DuaSourceType.quran => 'من القرآن',
        DuaSourceType.hadith => 'من السنة',
        DuaSourceType.mathur => 'مأثور',
        DuaSourceType.generalPermissible => 'دعاء عام جائز',
        DuaSourceType.userCreated => 'من أدعيتي',
        DuaSourceType.aiAssisted => 'صياغة مساعدة غير موثقة',
      };

  bool get isDocumented =>
      this == DuaSourceType.quran ||
      this == DuaSourceType.hadith ||
      this == DuaSourceType.mathur;

  /// تنبيه يظهر في التفاصيل لغير الموثّق.
  String? get disclaimer => switch (this) {
        DuaSourceType.generalPermissible =>
          'هذه صيغة عامة جائزة، وليست نصًا مأثورًا عن النبي ﷺ. للمسلم أن يدعو بما شاء من خير.',
        DuaSourceType.userCreated => 'هذا دعاء كتبتَه أنت، وليس من المحتوى الموثّق في التطبيق.',
        DuaSourceType.aiAssisted =>
          'صياغة مساعدة غير موثقة، ولا تُنسب لمصدر شرعي.',
        _ => null,
      };
}

T _byName<T extends Enum>(List<T> values, String? name, T fallback) {
  for (final v in values) {
    if (v.name == name) return v;
  }
  return fallback;
}

class Dua {
  const Dua({
    required this.id,
    required this.title,
    required this.arabicText,
    required this.category,
    required this.sourceType,
    required this.reference,
    this.stages = const [],
    this.transliteration,
    this.translations = const {},
    this.repeatCount = 1,
    this.grade,
    this.note,
    this.needsReview = false,
    this.isUserCreated = false,
    this.syncStatus = SyncStatus.localOnly,
    this.updatedAt,
  });

  final String id;
  final String title;
  final String arabicText;
  final DuaCategory category;
  final DuaSourceType sourceType;

  /// المرجع (قرآن/كتاب حديث/…). بدون أرقام أحاديث غير مؤكدة.
  final String reference;

  /// مراحل المناسك المرتبطة. الدعاء قد يناسب أكثر من مرحلة.
  final List<RitualStage> stages;
  final String? transliteration;

  /// مفتاح اللغة -> المعنى. ('en' حالياً)
  final Map<String, String> translations;
  final int repeatCount;
  final String? grade;
  final String? note;

  /// علم داخلي للمراجعة الشرعية قبل الإطلاق (يظهر في Debug فقط).
  final bool needsReview;
  final bool isUserCreated;
  final SyncStatus syncStatus;
  final DateTime? updatedAt;

  RitualStage get ritualStage => stages.isEmpty ? RitualStage.none : stages.first;

  String get copyText => '$arabicText\n\n— $title\n$reference\n(رفقة)';

  Dua copyWith({
    String? title,
    String? arabicText,
    DuaCategory? category,
    SyncStatus? syncStatus,
    DateTime? updatedAt,
  }) =>
      Dua(
        id: id,
        title: title ?? this.title,
        arabicText: arabicText ?? this.arabicText,
        category: category ?? this.category,
        sourceType: sourceType,
        reference: reference,
        stages: stages,
        transliteration: transliteration,
        translations: translations,
        repeatCount: repeatCount,
        grade: grade,
        note: note,
        needsReview: needsReview,
        isUserCreated: isUserCreated,
        syncStatus: syncStatus ?? this.syncStatus,
        updatedAt: updatedAt ?? this.updatedAt,
      );

  Map<String, dynamic> toJson() => {
        'id': id,
        'title': title,
        'arabicText': arabicText,
        'category': category.name,
        'sourceType': sourceType.name,
        'reference': reference,
        'stages': stages.map((s) => s.name).toList(),
        'transliteration': transliteration,
        'translations': translations,
        'repeatCount': repeatCount,
        'grade': grade,
        'note': note,
        'isUserCreated': isUserCreated,
        'syncStatus': syncStatus.name,
        'updatedAt': updatedAt?.toIso8601String(),
      };

  factory Dua.fromJson(Map<String, dynamic> j) => Dua(
        id: j['id'] as String,
        title: j['title'] as String? ?? '',
        arabicText: j['arabicText'] as String? ?? '',
        category: _byName(DuaCategory.values, j['category'] as String?, DuaCategory.general),
        sourceType:
            _byName(DuaSourceType.values, j['sourceType'] as String?, DuaSourceType.userCreated),
        reference: j['reference'] as String? ?? '',
        stages: ((j['stages'] as List?) ?? const [])
            .map((s) => _byName(RitualStage.values, s as String?, RitualStage.none))
            .toList(),
        transliteration: j['transliteration'] as String?,
        translations: Map<String, String>.from((j['translations'] as Map?) ?? const {}),
        repeatCount: (j['repeatCount'] as int?) ?? 1,
        grade: j['grade'] as String?,
        note: j['note'] as String?,
        isUserCreated: (j['isUserCreated'] as bool?) ?? true,
        syncStatus: _byName(SyncStatus.values, j['syncStatus'] as String?, SyncStatus.localOnly),
        updatedAt: DateTime.tryParse(j['updatedAt'] as String? ?? ''),
      );
}

import 'package:flutter/foundation.dart';

import '../../models/dua_model.dart';
import '../../screens/duas/duas_data.dart';
import 'duas_local_storage.dart';

/// الطبقة الوحيدة اللي الشاشات بتكلمها.
/// النهارده: محتوى مدمج + تخزين محلي. بعدين: يضاف Backend/AI هنا من غير لمس الواجهة.
class DuasRepository extends ChangeNotifier {
  DuasRepository({DuasLocalStorage? storage}) : _storage = storage ?? DuasLocalStorage();

  final DuasLocalStorage _storage;

  List<Dua> _user = [];
  Set<String> _favorites = {};
  double _fontScale = 1.0;
  bool _showTransliteration = true;
  bool _showTranslation = true;
  bool _ready = false;

  bool get isReady => _ready;
  double get fontScale => _fontScale;
  bool get showTransliteration => _showTransliteration;
  bool get showTranslation => _showTranslation;

  Future<void> init() async {
    _user = await _storage.loadUserDuas();
    _favorites = await _storage.loadFavorites();
    _fontScale = await _storage.loadFontScale();
    _showTransliteration = await _storage.loadShowTransliteration();
    _showTranslation = await _storage.loadShowTranslation();
    _ready = true;
    notifyListeners();
  }

  // ───── القراءة ─────
  List<Dua> get documented => DuasData.all;
  List<Dua> get userDuas => List.unmodifiable(_user);
  List<Dua> get all => [...DuasData.all, ..._user];

  Dua? byId(String id) {
    for (final d in all) {
      if (d.id == id) return d;
    }
    return null;
  }

  int countIn(DuaCategory c) => documented.where((d) => d.category == c).length;

  List<Dua> get favorites => all.where((d) => _favorites.contains(d.id)).toList();

  /// أدعية مرحلة معيّنة (للمرحلة 3: «الدعاء المناسب دلوقتي»).
  List<Dua> duasForStage(RitualStage stage) =>
      all.where((d) => d.stages.contains(stage)).toList();

  /// «دعاء اليوم»: ثابت طوال اليوم، من المحتوى الموثّق فقط.
  Dua duaOfTheDay([DateTime? now]) {
    final t = now ?? DateTime.now();
    final dayOfYear = t.difference(DateTime(t.year, 1, 1)).inDays;
    final pool = documented.where((d) => d.sourceType.isDocumented).toList();
    return pool[dayOfYear % pool.length];
  }

  // ───── البحث ─────
  static String normalizeArabic(String s) => s
      .replaceAll(RegExp(r'[\u064B-\u065F\u0670\u0640]'), '')
      .replaceAll(RegExp(r'[أإآٱ]'), 'ا')
      .replaceAll('ى', 'ي')
      .replaceAll('ة', 'ه')
      .toLowerCase();

  List<Dua> search({
    String query = '',
    DuaCategory? category,
    Set<DuaSourceType>? sources,
    bool favoritesOnly = false,
    bool mineOnly = false,
  }) {
    final q = normalizeArabic(query.trim());
    Iterable<Dua> r = mineOnly ? _user : all;
    if (favoritesOnly) r = r.where((d) => _favorites.contains(d.id));
    if (category != null) r = r.where((d) => d.category == category);
    if (sources != null) r = r.where((d) => sources.contains(d.sourceType));
    if (q.isNotEmpty) {
      r = r.where((d) {
        final hay = normalizeArabic([
          d.title,
          d.arabicText,
          d.reference,
          d.transliteration ?? '',
          ...d.translations.values,
        ].join(' '));
        return hay.contains(q);
      });
    }
    return r.toList();
  }

  // ───── المفضلة ─────
  bool isFavorite(String id) => _favorites.contains(id);

  Future<void> toggleFavorite(String id) async {
    _favorites.contains(id) ? _favorites.remove(id) : _favorites.add(id);
    notifyListeners();
    await _storage.saveFavorites(_favorites);
  }

  // ───── إعدادات القراءة ─────
  Future<void> setFontScale(double v, {bool persist = true}) async {
    _fontScale = v.clamp(0.8, 2.0);
    notifyListeners();
    if (persist) await _storage.saveFontScale(_fontScale);
  }

  Future<void> setShowTransliteration(bool v) async {
    _showTransliteration = v;
    notifyListeners();
    await _storage.saveShowTransliteration(v);
  }

  Future<void> setShowTranslation(bool v) async {
    _showTranslation = v;
    notifyListeners();
    await _storage.saveShowTranslation(v);
  }

  // ───── «أدعيتي»: جاهز للمرحلة 2 (الواجهة لسه) ─────
  Future<Dua> addUserDua({
    required String title,
    required String arabicText,
    required DuaCategory category,
  }) async {
    final now = DateTime.now();
    final dua = Dua(
      id: 'user_${now.microsecondsSinceEpoch}',
      title: title.trim(),
      arabicText: arabicText.trim(),
      category: category,
      sourceType: DuaSourceType.userCreated,
      reference: 'دعاء أضافه المستخدم',
      isUserCreated: true,
      syncStatus: SyncStatus.pending,
      updatedAt: now,
    );
    _user = [dua, ..._user];
    notifyListeners();
    await _storage.saveUserDuas(_user);
    return dua;
  }

  Future<void> updateUserDua(String id,
      {String? title, String? arabicText, DuaCategory? category}) async {
    final i = _user.indexWhere((d) => d.id == id);
    if (i < 0) return;
    _user[i] = _user[i].copyWith(
      title: title?.trim(),
      arabicText: arabicText?.trim(),
      category: category,
      syncStatus: SyncStatus.pending,
      updatedAt: DateTime.now(),
    );
    notifyListeners();
    await _storage.saveUserDuas(_user);
  }

  Future<void> deleteUserDua(String id) async {
    _user = _user.where((d) => d.id != id).toList();
    _favorites.remove(id);
    notifyListeners();
    await _storage.saveUserDuas(_user);
    await _storage.saveFavorites(_favorites);
  }
}

import 'dart:convert';

import 'package:shared_preferences/shared_preferences.dart';

import '../../models/dua_model.dart';

/// تخزين محلي بـSharedPreferences.
/// المحتوى الموثّق لا يُخزَّن هنا (موجود في duas_data.dart)، فقط بيانات المستخدم.
class DuasLocalStorage {
  static const _kUserDuas = 'rifqa.duas.user.v1';
  static const _kFavorites = 'rifqa.duas.favorites.v1';
  static const _kFontScale = 'rifqa.duas.fontScale.v1';
  static const _kShowTranslit = 'rifqa.duas.showTranslit.v1';
  static const _kShowTranslation = 'rifqa.duas.showTranslation.v1';

  Future<SharedPreferences> get _prefs => SharedPreferences.getInstance();

  Future<List<Dua>> loadUserDuas() async {
    final raw = (await _prefs).getString(_kUserDuas);
    if (raw == null || raw.isEmpty) return [];
    try {
      final list = jsonDecode(raw) as List;
      return list
          .map((e) => Dua.fromJson(Map<String, dynamic>.from(e as Map)))
          .toList();
    } catch (_) {
      return []; // بيانات تالفة: لا نكسر التطبيق
    }
  }

  Future<void> saveUserDuas(List<Dua> duas) async {
    await (await _prefs).setString(_kUserDuas, jsonEncode(duas.map((d) => d.toJson()).toList()));
  }

  Future<Set<String>> loadFavorites() async =>
      ((await _prefs).getStringList(_kFavorites) ?? const []).toSet();

  Future<void> saveFavorites(Set<String> ids) async =>
      (await _prefs).setStringList(_kFavorites, ids.toList());

  Future<double> loadFontScale() async => (await _prefs).getDouble(_kFontScale) ?? 1.0;
  Future<void> saveFontScale(double v) async => (await _prefs).setDouble(_kFontScale, v);

  Future<bool> loadShowTransliteration() async => (await _prefs).getBool(_kShowTranslit) ?? true;
  Future<void> saveShowTransliteration(bool v) async => (await _prefs).setBool(_kShowTranslit, v);

  Future<bool> loadShowTranslation() async => (await _prefs).getBool(_kShowTranslation) ?? true;
  Future<void> saveShowTranslation(bool v) async => (await _prefs).setBool(_kShowTranslation, v);
}

import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';

import '../../models/dua_model.dart';
import '../../services/duas/duas_repository.dart';
import 'widgets/dua_card.dart';

class DuaDetailsScreen extends StatefulWidget {
  const DuaDetailsScreen({super.key, required this.repository, required this.duaId});
  final DuasRepository repository;
  final String duaId;

  @override
  State<DuaDetailsScreen> createState() => _DuaDetailsScreenState();
}

class _DuaDetailsScreenState extends State<DuaDetailsScreen> {
  int _count = 0;

  static const _langNames = {
    'en': 'English',
    'ur': 'اردو',
    'id': 'Indonesia',
    'tr': 'Türkçe',
  };

  Future<void> _copy(Dua d) async {
    await Clipboard.setData(ClipboardData(text: d.copyText));
    if (!mounted) return;
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(const SnackBar(content: Text('تم نسخ الدعاء')));
  }

  void _tap(Dua d) {
    setState(() => _count++);
    if (d.repeatCount > 1 && _count == d.repeatCount) {
      HapticFeedback.heavyImpact();
    } else {
      HapticFeedback.selectionClick();
    }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: ListenableBuilder(
        listenable: widget.repository,
        builder: (context, _) {
          final repo = widget.repository;
          final dua = repo.byId(widget.duaId);
          if (dua == null) {
            return const Scaffold(body: Center(child: Text('الدعاء غير موجود')));
          }
          final t = Theme.of(context);
          final fav = repo.isFavorite(dua.id);
          final disclaimer = dua.sourceType.disclaimer;

          return Scaffold(
            appBar: AppBar(
              title: Text(dua.title, style: const TextStyle(fontSize: 18)),
              actions: [
                IconButton(
                  tooltip: 'نسخ',
                  icon: const Icon(Icons.copy_rounded),
                  onPressed: () => _copy(dua),
                ),
                IconButton(
                  tooltip: fav ? 'إزالة من المفضلة' : 'إضافة للمفضلة',
                  icon: Icon(fav ? Icons.favorite : Icons.favorite_border,
                      color: fav ? Colors.redAccent : null),
                  onPressed: () => repo.toggleFavorite(dua.id),
                ),
              ],
            ),
            body: ListView(
              padding: const EdgeInsets.all(16),
              children: [
                Wrap(spacing: 8, runSpacing: 6, children: [
                  DuaSourceBadge(dua.sourceType),
                  Chip(
                    label: Text(dua.category.label),
                    visualDensity: VisualDensity.compact,
                  ),
                ]),
                const SizedBox(height: 12),

                // النص العربي
                Card(
                  child: Padding(
                    padding: const EdgeInsets.symmetric(horizontal: 18, vertical: 24),
                    child: SelectableText(
                      dua.arabicText,
                      textAlign: TextAlign.center,
                      style: t.textTheme.headlineSmall?.copyWith(
                        fontSize: 24 * repo.fontScale,
                        height: 2.1,
                        fontWeight: FontWeight.w500,
                      ),
                    ),
                  ),
                ),

                // التحكم في حجم الخط
                Row(children: [
                  const Icon(Icons.text_decrease, size: 20),
                  Expanded(
                    child: Slider(
                      min: 0.8,
                      max: 2.0,
                      divisions: 12,
                      value: repo.fontScale.clamp(0.8, 2.0),
                      onChanged: (v) => repo.setFontScale(v, persist: false),
                      onChangeEnd: (v) => repo.setFontScale(v),
                    ),
                  ),
                  const Icon(Icons.text_increase, size: 20),
                ]),

                // العداد
                _counter(dua, t),

                // النطق
                if (dua.transliteration != null) ...[
                  SwitchListTile(
                    contentPadding: EdgeInsets.zero,
                    title: const Text('عرض النطق بالحروف اللاتينية'),
                    value: repo.showTransliteration,
                    onChanged: repo.setShowTransliteration,
                  ),
                  if (repo.showTransliteration)
                    _section('النطق', dua.transliteration!, t, ltr: true),
                ],

                // المعنى
                if (dua.translations.isNotEmpty) ...[
                  SwitchListTile(
                    contentPadding: EdgeInsets.zero,
                    title: const Text('عرض المعنى'),
                    value: repo.showTranslation,
                    onChanged: repo.setShowTranslation,
                  ),
                  if (repo.showTranslation)
                    for (final e in dua.translations.entries)
                      _section('المعنى (${_langNames[e.key] ?? e.key})', e.value, t, ltr: e.key != 'ur'),
                ],

                // المصدر
                _sourceCard(dua, t),

                if (disclaimer != null)
                  Padding(
                    padding: const EdgeInsets.only(top: 10),
                    child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
                      const Icon(Icons.info_outline, size: 18),
                      const SizedBox(width: 8),
                      Expanded(child: Text(disclaimer, style: t.textTheme.bodySmall)),
                    ]),
                  ),

                if (kDebugMode && dua.needsReview)
                  Padding(
                    padding: const EdgeInsets.only(top: 10),
                    child: Text(
                      '⚠️ [Debug فقط] هذا المحتوى يحتاج مراجعة شرعية قبل الإطلاق.',
                      style: t.textTheme.bodySmall?.copyWith(color: t.colorScheme.error),
                    ),
                  ),
                const SizedBox(height: 24),
              ],
            ),
          );
        },
      ),
    );
  }

  Widget _counter(Dua d, ThemeData t) {
    final target = d.repeatCount;
    final done = target > 1 && _count >= target;
    return Card(
      color: done ? t.colorScheme.primaryContainer : null,
      child: InkWell(
        borderRadius: BorderRadius.circular(12),
        onTap: () => _tap(d),
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
          child: Row(children: [
            Text(
              target > 1 ? '$_count / $target' : '$_count',
              style: t.textTheme.headlineSmall?.copyWith(fontWeight: FontWeight.bold),
            ),
            const SizedBox(width: 12),
            Expanded(
              child: Text(done ? 'تم العدد المطلوب ✓' : 'اضغط للعدّ',
                  style: t.textTheme.bodyMedium),
            ),
            IconButton(
              tooltip: 'تصفير',
              icon: const Icon(Icons.refresh),
              onPressed: () => setState(() => _count = 0),
            ),
          ]),
        ),
      ),
    );
  }

  Widget _section(String title, String body, ThemeData t, {bool ltr = false}) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 10),
      child: Card(
        child: Padding(
          padding: const EdgeInsets.all(14),
          child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
            Text(title, style: t.textTheme.labelLarge?.copyWith(color: t.colorScheme.primary)),
            const SizedBox(height: 6),
            Directionality(
              textDirection: ltr ? TextDirection.ltr : TextDirection.rtl,
              child: Align(
                alignment: ltr ? Alignment.centerLeft : Alignment.centerRight,
                child: Text(body, style: t.textTheme.bodyMedium?.copyWith(height: 1.7)),
              ),
            ),
          ]),
        ),
      ),
    );
  }

  Widget _sourceCard(Dua d, ThemeData t) {
    return Card(
      child: Padding(
        padding: const EdgeInsets.all(14),
        child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
          Text('المصدر', style: t.textTheme.labelLarge?.copyWith(color: t.colorScheme.primary)),
          const SizedBox(height: 6),
          Text(d.reference, style: t.textTheme.bodyMedium?.copyWith(height: 1.7)),
          if (d.grade != null) ...[
